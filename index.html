        📖 Supabase 1 Dakikada Kurulum (SQL Tablosu)
        
Supabase panelinizde SQL Editor sayfasına girip şu kodu yapıştırıp RUN'a basın:

        
create table if not exists public.company_inventory (
  channel text primary key,
  products jsonb default '[]'::jsonb,
  suppliers jsonb default '[]'::jsonb,
  orders jsonb default '[]'::jsonb,
  sales_history jsonb default '[]'::jsonb,
  stockin_history jsonb default '[]'::jsonb,
  users jsonb default '[]'::jsonb,
  updated_at bigint,
  updated_by text
);
alter publication supabase_realtime add table public.company_inventory;
alter table public.company_inventory enable row level security;
create policy "Allow all" on public.company_inventory for all using (true) with check (true);
-- Mevcut tabloya users, sales_history, stockin_history eklemek icin:
alter table public.company_inventory add column if not exists sales_history jsonb default '[]'::jsonb;
alter table public.company_inventory add column if not exists stockin_history jsonb default '[]'::jsonb;
alter table public.company_inventory add column if not exists users jsonb default '[]'::jsonb;
